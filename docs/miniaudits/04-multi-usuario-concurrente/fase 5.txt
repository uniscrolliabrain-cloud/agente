```typescript
// PRESENCE_STORE_V1 — PresenceService sobre Store, para multi-proceso.
// TTL configurable (default 30s). GC periódico. Aislado por tenant.
// Si el despliegue es single-instance, presence.ts in-memory sigue válido.
import type { Store } from "../db.ts";

const KIND = "presence";

export interface PresenceEntry {
  userId: string;
  tenantId: string;
  resourceId: string;
  resourceType: "thread" | "project";
  lastSeen: number;
}

export class PresenceStore {
  constructor(
    private readonly db: Store,
    private readonly ttlMs: number,
  ) {}

  private key(tenantId: string, resourceType: string, resourceId: string): string {
    return `${tenantId}:${resourceType}:${resourceId}`;
  }

  async touch(
    tenantId: string,
    userId: string,
    resourceType: "thread" | "project",
    resourceId: string,
  ): Promise<{ others: Array<{ userId: string; lastSeen: number }> }> {
    const now = Date.now();
    const entry: PresenceEntry = { userId, tenantId, resourceId, resourceType, lastSeen: now };
    await this.db.put(tenantId, KIND, { ...entry, id: this.key(tenantId, resourceType, resourceId) + ":" + userId });
    const all = await this.db.list<PresenceEntry & { id: string }>(tenantId, KIND);
    const others: Array<{ userId: string; lastSeen: number }> = [];
    for (const e of all) {
      if (e.resourceType !== resourceType || e.resourceId !== resourceId) continue;
      if (e.userId === userId) continue;
      if (now - e.lastSeen <= this.ttlMs) {
        others.push({ userId: e.userId, lastSeen: e.lastSeen });
      } else {
        await this.db.remove(tenantId, KIND, e.id);
      }
    }
    return { others };
  }

  async leave(tenantId: string, userId: string, resourceType: "thread" | "project", resourceId: string): Promise<void> {
    const id = this.key(tenantId, resourceType, resourceId) + ":" + userId;
    await this.db.remove(tenantId, KIND, id);
  }

  async query(tenantId: string, resourceType: "thread" | "project", resourceId: string): Promise<Array<{ userId: string; lastSeen: number }>> {
    const now = Date.now();
    const all = await this.db.list<PresenceEntry & { id: string }>(tenantId, KIND);
    const result: Array<{ userId: string; lastSeen: number }> = [];
    for (const e of all) {
      if (e.resourceType !== resourceType || e.resourceId !== resourceId) continue;
      if (now - e.lastSeen <= this.ttlMs) {
        result.push({ userId: e.userId, lastSeen: e.lastSeen });
      }
    }
    return result;
  }

  async gc(tenantId: string): Promise<number> {
    const now = Date.now();
    const all = await this.db.list<PresenceEntry & { id: string }>(tenantId, KIND);
    let removed = 0;
    for (const e of all) {
      if (now - e.lastSeen > this.ttlMs) {
        await this.db.remove(tenantId, KIND, e.id);
        removed++;
      }
    }
    return removed;
  }
}
```

```typescript
// EDIT_LOCK_STORE_V1 — EditLockService sobre Store, para multi-proceso.
// TTL configurable (default 10s). Locks con caducidad. Aislado por tenant.
// Si el despliegue es single-instance, edit-lock.ts in-memory sigue válido.
import type { Store } from "../db.ts";

const KIND = "edit-lock";

export interface LockEntry {
  id: string;
  tenantId: string;
  userId: string;
  resourceType: string;
  resourceId: string;
  acquiredAt: number;
}

export class EditLockStore {
  constructor(
    private readonly db: Store,
    private readonly ttlMs: number,
  ) {}

  private key(tenantId: string, resourceType: string, resourceId: string): string {
    return `${tenantId}:${resourceType}:${resourceId}`;
  }

  async acquire(tenantId: string, userId: string, resourceType: string, resourceId: string): Promise<LockEntry | null> {
    const id = this.key(tenantId, resourceType, resourceId);
    const existing = await this.db.get<LockEntry>(tenantId, KIND, id);
    const now = Date.now();
    if (existing && now - existing.acquiredAt <= this.ttlMs && existing.userId !== userId) {
      return null;
    }
    const entry: LockEntry = { id, tenantId, userId, resourceType, resourceId, acquiredAt: now };
    await this.db.put(tenantId, KIND, entry);
    return entry;
  }

  async release(tenantId: string, userId: string, resourceType: string, resourceId: string): Promise<void> {
    const id = this.key(tenantId, resourceType, resourceId);
    const existing = await this.db.get<LockEntry>(tenantId, KIND, id);
    if (existing && existing.userId === userId) {
      await this.db.remove(tenantId, KIND, id);
    }
  }

  async gc(tenantId: string): Promise<number> {
    const now = Date.now();
    const all = await this.db.list<LockEntry>(tenantId, KIND);
    let removed = 0;
    for (const e of all) {
      if (now - e.acquiredAt > this.ttlMs) {
        await this.db.remove(tenantId, KIND, e.id);
        removed++;
      }
    }
    return removed;
  }
}
```

```typescript
// SSE_REPLAY_BUFFER_V1 — ReplayBuffer por actor para SSE Last-Event-ID.
// Guarda los últimos N eventos por actorId. Si el cliente reconecta con
// Last-Event-ID, recibe los eventos perdidos. Condicional a reconexión.
export interface ReplayEvent {
  id: string;
  actorId: string;
  type: string;
  data: unknown;
  at: number;
}

export class ReplayBuffer {
  private readonly byActor = new Map<string, ReplayEvent[]>();

  constructor(private readonly maxSize: number) {}

  push(event: ReplayEvent): void {
    let buffer = this.byActor.get(event.actorId);
    if (!buffer) {
      buffer = [];
      this.byActor.set(event.actorId, buffer);
    }
    buffer.push(event);
    if (buffer.length > this.maxSize) {
      buffer.shift();
    }
  }

  since(actorId: string, lastEventId: string): ReplayEvent[] {
    const buffer = this.byActor.get(actorId);
    if (!buffer) return [];
    const idx = buffer.findIndex((e) => e.id === lastEventId);
    if (idx === -1) return buffer.slice();
    return buffer.slice(idx + 1);
  }

  clear(actorId: string): void {
    this.byActor.delete(actorId);
  }
}
```
