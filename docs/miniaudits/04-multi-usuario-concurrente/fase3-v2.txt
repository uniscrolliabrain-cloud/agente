// 2.1: sin z.any(). Tipos reales.
// 2.2: toAssignment produce vista normalizada; applyAssignment aplica solo
// campos autorizados; los campos específicos se conservan en el record original.
import type { AssignmentAdapter } from "../registry.ts";
import type { Assignment, AssignmentStatus } from "../../../../../../packages/domain/src/assignment.ts";
import type { Goal } from "../../../../../../packages/domain/src/agent.ts";

const KIND = "goal" as const;

export class GoalAdapter implements AssignmentAdapter<Goal> {
  readonly kind = KIND;

  toAssignment(record: Goal): Assignment {
    return {
      // B-N1: id = `${kind}:${recordId}`.
      id: `${KIND}:${record.id}`,
      tenantId: (record as { tenantId?: string }).tenantId ?? "default",
      kind: KIND,
      assignedTo: (record as { assignedTo?: string }).assignedTo ?? null,
      assignedBy: (record as { assignedBy?: string }).assignedBy ?? null,
      // B-N2: status normalizado + nativeStatus.
      status: this.toCanonicalStatus(record.status),
      nativeStatus: record.status,
      evidence: [],
      createdAt: record.createdAt,
      updatedAt: record.createdAt,
    };
  }

  applyAssignment(record: Goal, changes: Partial<Assignment>): Goal {
    // 2.2: solo campos autorizados. Los específicos del Goal se conservan.
    const next: Goal = { ...record };
    if (changes.assignedTo !== undefined) {
      (next as { assignedTo?: string | null }).assignedTo = changes.assignedTo;
    }
    if (changes.assignedBy !== undefined) {
      (next as { assignedBy?: string | null }).assignedBy = changes.assignedBy;
    }
    return next;
  }

  applyStatus(record: Goal, status: AssignmentStatus): Goal {
    // B-N2: mapea el status canónico al estado nativo del Goal.
    const native = this.toNativeStatus(status);
    return { ...record, status: native };
  }

  private toCanonicalStatus(native: Goal["status"]): AssignmentStatus {
    switch (native) {
      case "active":
        return "active";
      case "paused":
        return "paused";
      case "completed":
        return "completed";
      default:
        return "active";
    }
  }

  private toNativeStatus(status: AssignmentStatus): Goal["status"] {
    switch (status) {
      case "active":
        return "active";
      case "paused":
        return "paused";
      case "completed":
      case "succeeded":
        return "completed";
      default:
        return "active";
    }
  }
}
