  const calls: Array<{ to: string; text: string; generation: string; apiKey: string }> = [];

test("AUDIT: rotating the channel during reply creation must reject rather than bind the new channel", async () => {
  const f = await fixture();
  await f.run(f.a, () => f.scoped.put(f.a.owner, "whatsapp-inbox", { id: "audit-inbox", tenantId: f.a.tenantId, owner: f.a.owner, provider: "evolution", accountId: f.a.tenantId, from: `${recipient}@s.whatsapp.net`, text: "Hola", providerMessageId: "inbound-audit", receivedAt: new Date().toISOString(), status: "queued" }));
  const original = f.scoped.get.bind(f.scoped); let reads = 0;
  f.scoped.get = async (...args: Parameters<typeof original>) => {
    if (args[1] === "whatsapp-connections" && ++reads === 2) await f.provision(f.a, { instance: "other-account" });
    return original(...args);
  };
  await assert.rejects(f.run(f.a, () => f.client().createReply(f.a, "audit-inbox", { text: "Respuesta" }, "audit-reply-key")), /canal|conexion|cambi/i);
});
